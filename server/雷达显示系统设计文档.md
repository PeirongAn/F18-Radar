# 雷达显示系统设计文档

## 概述

雷达显示系统是整个雷达威胁排序系统中的核心可视化组件，负责实时渲染雷达扫描界面、目标显示、跟踪信息以及各种雷达操作控制元素。系统采用 React + Konva.js 的技术方案，实现了高性能的 2D Canvas 渲染和复杂的雷达显示逻辑。

const renderDistanceRings = ({ centerX, centerY, radarRange, scale }: GridProps) => {
  const rings = []
  const ringCount = 4  // 4个距离圈
  
  for (let i = 1; i <= ringCount; i++) {
    const radius = (radarRange / ringCount) * i * scale
    rings.push(
      <Circle
        key={i}
        x={centerX}
        y={centerY}
        radius={radius}
        stroke="#00ff00"
        strokeWidth={1}
      />
    )
  }
  
  return rings
}

const renderBearingLines = ({ centerX, centerY, radarRange, scale }: GridProps) => {
  const lines = []
  const bearingStep = 30  // 每30°一条方位线
  
  for (let bearing = 0; bearing < 360; bearing += bearingStep) {
    const angleRad = bearing * Math.PI / 180
    const endX = centerX + radarRange * scale * Math.cos(angleRad)
    const endY = centerY + radarRange * scale * Math.sin(angleRad)
    
    lines.push(
      <Line
        key={bearing}
        points={[centerX, centerY, endX, endY]}
        stroke="#00ff00"
        strokeWidth={1}
        opacity={0.5}
      />
    )
  }
  
  return lines
}
```

interface ScanAnimation {

const updateScanPosition = useCallback((currentTime: number) => {
  const deltaTime = currentTime - animRef.current.lastTime
  const speed = BASE_SPEED * scanSpeed  // 像素/毫秒
  const distance = speed * deltaTime
  
  if (animRef.current.isForward) {
    animRef.current.position += distance
    if (animRef.current.position >= rangeRef.current.end) {
      animRef.current.position = rangeRef.current.end
      animRef.current.isForward = false
      onScanCycleComplete?.()  // 扫描完成回调
    }
  } else {
    animRef.current.position -= distance
    if (animRef.current.position <= rangeRef.current.start) {
      animRef.current.position = rangeRef.current.start
      animRef.current.isForward = true
    }
  }
  
  setDisplayPosition(animRef.current.position)
  animRef.current.lastTime = currentTime
}, [scanSpeed, onScanCycleComplete])
```

const calculateScanRange = (scanAngle: number, centerX: number, boxWidth: number) => {
  const scanCenterX = centerX + (centerOffset || 0)
  
  switch (scanAngle) {
    case 60:  // 全范围扫描
      return {
        start: framePositions.startX,
        end: framePositions.endX

      }

  }

interface ClickDetection {
  target: TargetData
  clickPoint: Point
  tolerance: number         // 点击容忍度（像素）
}

const isTargetClicked = ({ target, clickPoint, tolerance }: ClickDetection): boolean => {
  const distance = Math.sqrt(
    Math.pow(clickPoint.x - target.x, 2) + 
    Math.pow(clickPoint.y - target.y, 2)
  )
  
  return distance <= tolerance
}

const handleTargetClick = useCallback((clickPoint: Point) => {
  const tolerance = 15  // 15像素容忍度

class TargetObjectPool {
  private pool: TargetObject[] = []
  private active: Map<string, TargetObject> = new Map()
  
  acquire(targetId: string): TargetObject {
    let obj = this.pool.pop()
    if (!obj) {
      obj = new TargetObject()
    }
    obj.reset(targetId)
    this.active.set(targetId, obj)
    return obj
  }
  
  release(targetId: string) {
    const obj = this.active.get(targetId)
    if (obj) {
      this.active.delete(targetId)
      this.pool.push(obj)
    }
  }
}

const calculateResponsiveLayout = (
  containerWidth: number, 
  containerHeight: number
): LayoutConfig => {
  const aspectRatio = 4 / 3  // 标准雷达显示比例
  
  let displayWidth, displayHeight
  
  if (containerWidth / containerHeight > aspectRatio) {
    // 容器更宽，以高度为准
    displayHeight = containerHeight * 0.9
    displayWidth = displayHeight * aspectRatio
  } else {
    // 容器更高，以宽度为准
    displayWidth = containerWidth * 0.9
    displayHeight = displayWidth / aspectRatio
  }
  
  return {
    width: displayWidth,
    height: displayHeight,
    centerX: displayWidth / 2,
    centerY: displayHeight / 2,
    scale: displayWidth / 1000  // 基准缩放
  }
}

  const [radarData, setRadarData] = useState<RadarData | null>(null)
  
  useEffect(() => {
    const handleWebSocketMessage = (data: any) => {
      switch (data.type) {
        case 'radar_update':
          setRadarData(prev => ({
            ...prev,
            targets: data.targets,
            timestamp: data.timestamp
          }))
          break
          
        case 'target_update':
          setRadarData(prev => {
            if (!prev) return null
            
            const updatedTargets = prev.targets.map(target =>
              target.id === data.targetId 
                ? { ...target, ...data.updates }
                : target
            )
            
            return { ...prev, targets: updatedTargets }
          })
          break
      }
    }
    
    // WebSocket 监听器注册
    globalWS.subscribe(handleWebSocketMessage)
    
    return () => globalWS.unsubscribe(handleWebSocketMessage)
  }, [])
  
  return { radarData }
}

class PerformanceMonitor {
  private frameCount = 0
  private lastTime = 0
  private fps = 0
  
  startMonitoring() {
    const measure = (currentTime: number) => {
      this.frameCount++
      
      if (currentTime - this.lastTime >= 1000) {
        this.fps = this.frameCount
        this.frameCount = 0
        this.lastTime = currentTime
        
        // 性能警告
        if (this.fps < 30) {
          console.warn(`低帧率警告: ${this.fps} FPS`)
        }
      }
      
      requestAnimationFrame(measure)
    }
    
    requestAnimationFrame(measure)
  }
  
  getCurrentFPS(): number {
    return this.fps
  }
}
