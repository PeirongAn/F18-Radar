# 雷达威胁排序系统工程设计说明文档

## 概述

雷达威胁排序系统是一个基于Web技术的智能空中对抗等效试验系统，采用前后端分离架构，集成了AI智能体技术，实现了雷达目标跟踪和威胁态势感知的交互式训练平台。系统支持传感器交互、威胁排序、传感器信任和威胁排序信任四个核心任务子系统。

class AIThreatAssessment:
    """AI威胁评估"""
    
    def assess_threat_priority(self, threats: List[Threat]) -> List[Threat]:
        """AI威胁优先级评估"""
        scored_threats = []
        
        for threat in threats:
            score = self.calculate_threat_score(threat)
            threat.ai_assessment_score = score
            scored_threats.append(threat)
        
        # 按AI评分排序
        return sorted(scored_threats, key=lambda t: t.ai_assessment_score, reverse=True)
    
    def calculate_threat_score(self, threat: Threat) -> float:
        """计算威胁评分"""
        base_score = threat.base_priority
        
        # 距离因素
        distance_factor = 1 / max(threat.distance, 1)
        
        # 类型因素
        type_factor = self.get_threat_type_weight(threat.type)
        
        # 武器系统因素
        weapon_factor = self.get_weapon_system_weight(threat.weapon_systems)
        
        return base_score * distance_factor * type_factor * weapon_factor

**目标跟踪算法**:

```python
class TargetTracker:
    """目标跟踪器"""
    
    def __init__(self):
        self.tracked_targets = {}
        self.target_history = {}
    
    def update_target_track(self, target_id: str, position: Tuple[float, float], 
                           timestamp: float) -> TrackInfo:
        """更新目标轨迹"""
        if target_id not in self.target_history:
            self.target_history[target_id] = []
        
        # 添加新的位置点
        self.target_history[target_id].append({
            'position': position,
            'timestamp': timestamp
        })
        
        # 保持历史记录长度
        if len(self.target_history[target_id]) > 10:
            self.target_history[target_id] = self.target_history[target_id][-10:]
        
        # 计算速度和方向
        velocity = self.calculate_velocity(target_id)
        heading = self.calculate_heading(target_id)
        
        return TrackInfo(
            target_id=target_id,
            current_position=position,
            velocity=velocity,
            heading=heading,
            track_quality=self.assess_track_quality(target_id)
        )
    
    def calculate_velocity(self, target_id: str) -> float:
        """计算目标速度"""
        history = self.target_history[target_id]
        if len(history) < 2:
            return 0.0
        
        recent = history[-2:]
        dt = recent[1]['timestamp'] - recent[0]['timestamp']
        dx = recent[1]['position'][0] - recent[0]['position'][0]
        dy = recent[1]['position'][1] - recent[0]['position'][1]
        
        distance = math.sqrt(dx*dx + dy*dy)
        return distance / dt if dt > 0 else 0.0
```

class PriorityCalculator:
    """威胁优先级计算器"""
    
    def __init__(self):
        self.weight_config = {
            'distance': 0.3,
            'threat_type': 0.25,
            'weapon_capability': 0.25,
            'tactical_importance': 0.2
        }
    
    def calculate_threat_priorities(self, threats: List[EnhancedThreat]) -> List[EnhancedThreat]:
        """计算威胁优先级"""
        for threat in threats:
            threat.priority_score = self.calculate_individual_priority(threat)
        
        # 按优先级排序
        sorted_threats = sorted(threats, key=lambda t: t.priority_score, reverse=True)
        
        # 分配优先级等级
        for i, threat in enumerate(sorted_threats):
            if i < len(sorted_threats) * 0.3:
                threat.priority_level = 'high'
            elif i < len(sorted_threats) * 0.7:
                threat.priority_level = 'medium'
            else:
                threat.priority_level = 'low'
        
        return sorted_threats
    
    def calculate_individual_priority(self, threat: EnhancedThreat) -> float:
        """计算单个威胁的优先级分数"""
        # 距离因子（越近优先级越高）
        distance_score = 1.0 / max(threat.distance, 1.0)
        
        # 威胁类型因子
        type_score = self.get_threat_type_score(threat.type)
        
        # 武器能力因子
        weapon_score = self.get_weapon_capability_score(threat)
        
        # 战术重要性因子
        tactical_score = self.get_tactical_importance_score(threat)
        
        # 加权计算
        total_score = (
            distance_score * self.weight_config['distance'] +
            type_score * self.weight_config['threat_type'] +
            weapon_score * self.weight_config['weapon_capability'] +
            tactical_score * self.weight_config['tactical_importance']
        )
        
        return total_score
    
    def get_threat_type_score(self, threat_type: str) -> float:
        """获取威胁类型评分"""
        type_scores = {
            'PrimaryAir': 1.0,      # 空中主要威胁
            'SecondaryAir': 0.8,    # 空中次要威胁
            'PrimaryAntiAir': 0.9,  # 防空主要威胁

    """威胁评估模型"""

            score = factor.evaluate(threat, context)
            factor_scores[factor_name] = score

        return ThreatAssessment(

            factor_scores=factor_scores,
            assessment_time=time.time()
        )
    
    def calculate_overall_score(self, factor_scores: Dict[str, float]) -> float:
        """计算综合评分"""
        weights = {
            'kinematic': 0.25,    # 运动学因素
            'capability': 0.35,   # 能力因素
            'intent': 0.25,       # 意图因素
            'opportunity': 0.15   # 机会因素
        }
        
        weighted_sum = sum(
            factor_scores[factor] * weights[factor]
            for factor in factor_scores
        )
        
        return min(max(weighted_sum, 0.0), 1.0)  # 限制在[0,1]范围

            'trust_level': self.current_trust_level,

        })
        
        return self.current_trust_level

        """计算信任评分"""

            'accuracy': 0.4,      # 准确性权重最高
            'reliability': 0.3,   # 可靠性
            'consistency': 0.2,   # 一致性
            'timeliness': 0.1     # 及时性

class AdaptiveTrustCalibration:
    """自适应信任校准"""
    
    def __init__(self):
        self.calibration_history = []
        self.user_feedback_model = UserFeedbackModel()
        self.performance_tracker = PerformanceTracker()
    
    def calibrate_trust(self, predicted_trust: float, 
                       actual_performance: float,
                       user_feedback: Optional[UserFeedback] = None) -> float:
        """校准信任度"""
        # 基于历史表现校准
        performance_factor = self.calculate_performance_factor()
        
        # 基于用户反馈校准
        feedback_factor = 1.0
        if user_feedback:
            feedback_factor = self.user_feedback_model.process_feedback(user_feedback)
        
        # 基于环境复杂度校准
        complexity_factor = self.calculate_complexity_factor()
        
        # 综合校准
        calibrated_trust = predicted_trust * performance_factor * feedback_factor * complexity_factor
        
        # 记录校准历史
        self.calibration_history.append({
            'timestamp': time.time(),
            'predicted_trust': predicted_trust,
            'calibrated_trust': calibrated_trust,
            'actual_performance': actual_performance,
            'calibration_factors': {
                'performance': performance_factor,
                'feedback': feedback_factor,
                'complexity': complexity_factor
            }
        })
        
        return max(0.0, min(1.0, calibrated_trust))
    
    def calculate_performance_factor(self) -> float:
        """计算性能校准因子"""
        recent_performance = self.performance_tracker.get_recent_performance()
        if not recent_performance:
            return 1.0
        
        # 基于最近的性能表现调整信任度
        avg_performance = sum(recent_performance) / len(recent_performance)
        
        if avg_performance > 0.9:
            return 1.1    # 提升信任
        elif avg_performance < 0.6:
            return 0.8    # 降低信任
        else:
            return 1.0    # 保持不变

      {/* 信任趋势分析 */}
      <div className="trust-trends">

        <TrustFeedbackPanel

        />
      </div>
    </div>
  )
